cmake_minimum_required(VERSION 3.22)
project(SecureChromium VERSION 1.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_AUTOMOC ON)
include(CTest)
find_package(Qt6 6.4 REQUIRED COMPONENTS Widgets WebEngineWidgets)
add_library(securechromium_core src/policy.cpp src/browser.cpp)
target_include_directories(securechromium_core PUBLIC src)
target_link_libraries(securechromium_core PUBLIC Qt6::Widgets Qt6::WebEngineWidgets)
target_compile_options(securechromium_core PRIVATE $<$<CXX_COMPILER_ID:GNU,Clang>:-Wall;-Wextra;-Wpedantic;-Wconversion;-Wshadow>)
add_executable(securechromium src/main.cpp)
target_link_libraries(securechromium PRIVATE securechromium_core)
install(TARGETS securechromium RUNTIME DESTINATION bin BUNDLE DESTINATION .)
if(BUILD_TESTING)
  find_package(Qt6 REQUIRED COMPONENTS Test Network)
  add_executable(policy_tests tests/policy_tests.cpp)
  target_link_libraries(policy_tests PRIVATE securechromium_core Qt6::Test)
  add_dependencies(policy_tests securechromium)
  add_test(NAME policy COMMAND policy_tests)
  add_executable(browser_tests tests/browser_tests.cpp)
  target_link_libraries(browser_tests PRIVATE securechromium_core Qt6::Test Qt6::Network)
  add_test(NAME browser COMMAND browser_tests)
  set_tests_properties(browser PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen;QTWEBENGINE_CHROMIUM_FLAGS=--disable-gpu" TIMEOUT 60)
endif()
